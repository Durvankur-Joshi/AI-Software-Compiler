    def run(self, intent):

        prompt = f"""
You are a software architecture planner.

Convert the intent into STRICT JSON.

You MUST return ONLY valid JSON.

Do NOT return markdown.
Do NOT return explanations.
Do NOT add extra keys.

Required JSON structure:

{{
  "entities": [

        parsed = json.loads(response)

        validated = ArchitectureSchema(
            **parsed
        )

        return validated

from app.pipeline.intent_extractor import IntentExtractor

from app.pipeline.architecture_planner import ArchitecturePlanner

from app.pipeline.schema_generator import SchemaGenerator

from app.pipeline.validator import SchemaValidator

from app.pipeline.repair_engine import RepairEngine

from app.runtime.sqlite_runtime import SQLiteRuntime

from app.services.supabase_service import supabase


class PipelineOrchestrator:

    def __init__(self):

        self.intent_extractor = IntentExtractor()

        self.architecture_planner = ArchitecturePlanner()

        self.schema_generator = SchemaGenerator()

        self.validator = SchemaValidator()

        self.repair_engine = RepairEngine()

        self.runtime = SQLiteRuntime()

    def run(self, user_prompt: str):

        # =========================
        # 1. INTENT EXTRACTION
        # =========================

        intent = self.intent_extractor.run(
            user_prompt
        )

        # =========================
        # 2. ARCHITECTURE PLANNING
        # =========================

        architecture = self.architecture_planner.run(
            intent
        )

        # =========================
        # 3. SCHEMA GENERATION
        # =========================

        schemas = self.schema_generator.generate_all_schemas(
            architecture
        )

        db_schema = schemas["database_schema"]

        api_schema = schemas["api_schema"]

        auth_schema = schemas["auth_schema"]

        # Generate UI separately
        ui_schema = self.schema_generator.generate_ui_schema(
            architecture
        )

        # =========================
        # 4. VALIDATION
        # =========================

        validation_errors = self.validator.validate(
            database_schema=db_schema,
            api_schema=api_schema,
            ui_schema=ui_schema,
            auth_schema=auth_schema
        )

        # =========================
        # 5. REPAIR ENGINE
        # =========================

        repaired = False

        if validation_errors:

            repaired = True

            api_schema = self.repair_engine.repair_api_schema(
                api_schema,
                db_schema
            )

            # Re-run validation after repair
            validation_errors = self.validator.validate(
                database_schema=db_schema,
                api_schema=api_schema,
                ui_schema=ui_schema,
                auth_schema=auth_schema
            )

        # =========================
        # 6. RUNTIME EXECUTION
        # =========================

        runtime_status = "success"

        try:

            self.runtime.create_database(
                db_schema
            )

        except Exception as e:

            runtime_status = f"failed: {str(e)}"

        # =========================
        # 7. SAVE TO SUPABASE
        # =========================

        supabase.table(
            "generated_apps"
        ).insert({

            "prompt": user_prompt,

            "intent": intent.model_dump(),

            "architecture": architecture.model_dump(),

            "database_schema": db_schema.model_dump(),

            "api_schema": api_schema.model_dump(),

            "ui_schema": ui_schema.model_dump(),

            "auth_schema": auth_schema.model_dump(),

            "validation_errors": validation_errors,

            "repair_applied": repaired

        }).execute()

        # =========================
        # 8. FINAL RESPONSE
        # =========================

        return {

            "intent": intent.model_dump(),

            "architecture": architecture.model_dump(),

            "database": db_schema.model_dump(),

            "api": api_schema.model_dump(),

            "ui": ui_schema.model_dump(),

            "auth": auth_schema.model_dump(),

            "validation_errors": validation_errors,

            "repair_applied": repaired,

            "runtime_status": runtime_status
        }

class RepairEngine:

    def repair_api_schema(
        self,
        api_schema,
        db_schema
    ):

        valid_fields = set()

        for table in db_schema.tables:
            for column in table.columns:
                valid_fields.add(column.name)

        for endpoint in api_schema.endpoints:

            endpoint.request_fields = [
                field
                for field in endpoint.request_fields
                if field in valid_fields
            ]

        return api_schema

class SchemaValidator:

    def validate(
        self,
        database_schema,
        api_schema,
        ui_schema,
        auth_schema
    ):

        errors = []

        errors.extend(
            self.validate_database(
                database_schema
            )
        )

        errors.extend(
            self.validate_api_vs_db(
                database_schema,
                api_schema
            )
        )

        errors.extend(
            self.validate_auth_vs_api(
                api_schema,
                auth_schema
            )
        )

        errors.extend(
            self.validate_ui_vs_api(
                ui_schema,
                api_schema
            )
        )

        return errors

    def validate_database(
        self,
        database_schema
    ):

        errors = []

        for table in database_schema.tables:

            column_names = [
                column.name
                for column in table.columns
            ]

            if "id" not in column_names:

                errors.append(
                    f"{table.table_name} missing id field"
                )

        return errors

    def validate_api_vs_db(
        self,
        database_schema,
        api_schema
    ):

        errors = []

        db_fields = set()

        for table in database_schema.tables:

            for column in table.columns:

                db_fields.add(
                    column.name
                )

        for endpoint in api_schema.endpoints:

            for field in endpoint.request_fields:

                if (
                    field not in db_fields
                    and field != "password"
                ):

                    errors.append(
                        f"API request field '{field}' not found in DB schema"
                    )

            for field in endpoint.response_fields:

                if (
                    field not in db_fields
                    and field not in ["token", "message", "data"]
                ):

                    errors.append(
                        f"API response field '{field}' not found in DB schema"
                    )

        return errors

    def validate_auth_vs_api(
        self,
        api_schema,
        auth_schema
    ):

        errors = []

        api_routes = set()

        for endpoint in api_schema.endpoints:

            api_routes.add(
                endpoint.path
            )

        for permission in auth_schema.permissions:

            for route in permission.allowed_routes:

                if route not in api_routes:

                    errors.append(
                        f"Auth route '{route}' does not exist in API schema"
                    )

        return errors

    def validate_ui_vs_api(
        self,
        ui_schema,
        api_schema
    ):

        errors = []

        api_routes = [
            endpoint.path
            for endpoint in api_schema.endpoints
        ]

        has_contacts_api = any(
            "/contacts" in route
            for route in api_routes
        )

        for page in ui_schema.pages:

            if (
                "Contact" in page.name
                and not has_contacts_api
            ):

                errors.append(
                    "UI references contacts but contacts API missing"
                )

        return errors

import sqlite3
import os


class SQLiteRuntime:

    def create_database(self, db_schema):

        os.makedirs(
            "generated",
            exist_ok=True
        )

        connection = sqlite3.connect(
            "generated/generated_app.db"
        )

        cursor = connection.cursor()

        for table in db_schema.tables:

            columns = []

            for column in table.columns:

                # Prevent duplicate id column
                if column.name.lower() == "id":
                    continue

                sql_type = column.type.upper()

                if sql_type not in [
                    "TEXT",
                    "INTEGER",
                    "BOOLEAN"
                ]:
                    sql_type = "TEXT"

                columns.append(
                    f"{column.name} {sql_type}"
                )

            query = f"""
            CREATE TABLE IF NOT EXISTS
            {table.table_name}
            (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                {",".join(columns)}
            )
            """

            print("\nSQL QUERY:\n")
            print(query)

            cursor.execute(query)

        connection.commit()

        connection.close()

import time

import google.generativeai as genai

from app.core.config import settings


genai.configure(
    api_key=settings.GEMINI_API_KEY
)


class GeminiService:

    def __init__(self):

        self.model = genai.GenerativeModel(
            settings.MODEL_NAME
        )

    def generate_json(
        self,
        prompt: str,
        retries: int = 3
    ):

        attempt = 0

        while attempt < retries:

            try:

                response = self.model.generate_content(
                    prompt,
                    generation_config={
                        "temperature": 0,
                        "response_mime_type": "application/json"
                    }
                )

                return response.text

            except Exception as e:

                print("\nGEMINI ERROR:\n")
                print(str(e))

                attempt += 1

                if attempt >= retries:
                    raise Exception(
                        f"Gemini failed after {retries} retries"
                    )

                wait_time = 5 * attempt

                print(
                    f"\nRetrying in {wait_time} seconds...\n"
                )

                time.sleep(wait_time)


gemini_service = GeminiService()
